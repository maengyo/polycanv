# 보여줄 것: 도구를 골라 열고, 원하는 자리에 놓고, 크기를 바꾼다.
# 좌표는 칸 단위(0 부터)이고 100x28 화면 기준이다.
#
# 첫 터미널은 (0,0) 에 48x14 로 열린다. 다음 것은 (3,3) 으로 어긋나 열린다.

wait 1.5
type ls
key enter
wait 1.0

# 도구 목록에서 하나 더 연다
key ctrl+b
key n
wait 1.5
key enter
wait 1.5

# 새로 열린 것(3,3)의 제목 줄을 잡아 오른쪽으로 옮긴다
drag 20,3 68,3
wait 0.6

type echo second session
key enter
wait 1.2

# 오른쪽 아래 모서리를 잡아 아래로 늘린다 (패널은 51,3 에 48x14)
drag 98,16 98,25
wait 2.0
